To investigate how Grafana's alert evaluation and notification system behaves under stress, I wrote a test that exercises three alert rules concurrently with different timing characteristics. One rule is slow and times out, one is fast, and one is flaky with intermittent failures. The test applies rule updates and deletions while evaluations are already falling behind to observe the runtime behavior.

```go
package schedule

import (
	"context"
	"errors"
	"fmt"
	"net/url"
	"sync"
	"sync/atomic"
	"testing"
	"time"

	"github.com/benbjohnson/clock"
	"github.com/grafana/grafana-plugin-sdk-go/backend"
	"github.com/grafana/grafana-plugin-sdk-go/data"
	"github.com/prometheus/client_golang/prometheus"
	"github.com/prometheus/client_golang/prometheus/testutil"
	"github.com/stretchr/testify/mock"
	"golang.org/x/sync/errgroup"

	"github.com/grafana/grafana/pkg/infra/log"
	"github.com/grafana/grafana/pkg/infra/tracing"
	"github.com/grafana/grafana/pkg/services/ngalert/api/tooling/definitions"
	"github.com/grafana/grafana/pkg/services/ngalert/eval"
	"github.com/grafana/grafana/pkg/services/ngalert/metrics"
	"github.com/grafana/grafana/pkg/services/ngalert/models"
	"github.com/grafana/grafana/pkg/services/ngalert/state"
	"github.com/grafana/grafana/pkg/services/ngalert/writer"
	"github.com/grafana/grafana/pkg/setting"
)

func TestAlertingSchedulerStressAndNormalLoadDemo(t *testing.T) {
	type scenarioSpec struct {
		name             string
		slowEvalDuration time.Duration
		fastEvalDuration time.Duration
		flakyEvalDuration time.Duration
		flakyFailOnCall  int
		slowTimeoutOnCall int
		applyRuleUpdate  bool
		deleteRuleMidway bool
		ticks            int
	}

	scenarios := []scenarioSpec{
		{
			name:             "stress",
			slowEvalDuration: 1500 * time.Millisecond,
			fastEvalDuration: 200 * time.Millisecond,
			flakyEvalDuration: 700 * time.Millisecond,
			flakyFailOnCall:  2,
			slowTimeoutOnCall: 1,
			applyRuleUpdate:  true,
			deleteRuleMidway: true,
			ticks:            5,
		},
		{
			name:             "normal",
			slowEvalDuration: 200 * time.Millisecond,
			fastEvalDuration: 150 * time.Millisecond,
			flakyEvalDuration: 180 * time.Millisecond,
			flakyFailOnCall:  0,
			slowTimeoutOnCall: 0,
			applyRuleUpdate:  false,
			deleteRuleMidway: false,
			ticks:            4,
		},
	}

	for _, scenario := range scenarios {
		runScenario := func(name string, spec scenarioSpec, reg prometheus.Registerer) {
			testTracer := tracing.InitializeTracerForTest()
			testMetrics := metrics.NewNGAlert(reg)
			ctx, cancel := context.WithCancel(context.Background())
			defer cancel()
			dispatcherGroup, ctx := errgroup.WithContext(ctx)

			ruleStore := newFakeRulesStore()
			cfg := setting.UnifiedAlertingSettings{
				BaseInterval:            1 * time.Second,
				AdminConfigPollInterval: 10 * time.Minute,
			}

			const mainOrgID int64 = 1
			mockedClock := clock.NewMock()

			notifier := NewSyncAlertsSenderMock()
			notifier.EXPECT().Send(mock.Anything, mock.Anything, mock.Anything).Run(func(ctx context.Context, key models.AlertRuleKey, alerts definitions.PostableAlerts) {
				fmt.Printf("ALERT send uid=%s alerts=%d total=%d\n", key.UID, len(alerts.PostableAlerts), len(alerts.PostableAlerts))
			}).Return()

			appUrl := &url.URL{Scheme: "http", Host: "localhost"}

			var callCounts sync.Map
			var deletedRules sync.Map

			fakeEvaluator := &fakeConditionEvaluator{
				evalFunc: func(ctx context.Context, now time.Time, ruleUID string) (eval.Results, error) {
					countVal, _ := callCounts.LoadOrStore(ruleUID, new(int32))
					callNum := atomic.AddInt32(countVal.(*int32), 1)
					fmt.Printf("EVAL start uid=%s scheduled=%s call=%d\n", ruleUID, now.Format(time.RFC3339), callNum)

					var sleepDur time.Duration
					switch ruleUID {
					case "a-slow":
						sleepDur = spec.slowEvalDuration
						if spec.slowTimeoutOnCall > 0 && int(callNum) == spec.slowTimeoutOnCall {
							time.Sleep(sleepDur)
							fmt.Printf("EVAL error uid=%s scheduled=%s err=context deadline exceeded\n", ruleUID, now.Format(time.RFC3339))
							return nil, errors.New("context deadline exceeded")
						}
					case "b-fast":
						sleepDur = spec.fastEvalDuration
					case "c-flaky":
						sleepDur = spec.flakyEvalDuration
						if spec.flakyFailOnCall > 0 && int(callNum) == spec.flakyFailOnCall {
							time.Sleep(sleepDur)
							fmt.Printf("EVAL error uid=%s scheduled=%s err=upstream timeout\n", ruleUID, now.Format(time.RFC3339))
							return nil, errors.New("upstream timeout")
						}
					}

					select {
					case <-time.After(sleepDur):
					case <-ctx.Done():
						if _, deleted := deletedRules.Load(ruleUID); deleted {
							fmt.Printf("EVAL canceled uid=%s scheduled=%s err=context canceled\nrule deleted\n", ruleUID, now.Format(time.RFC3339))
						} else {
							fmt.Printf("EVAL canceled uid=%s scheduled=%s err=context canceled\n", ruleUID, now.Format(time.RFC3339))
						}
						return nil, ctx.Err()
					}

					instanceLabel := ruleUID[:1] + "-instance"
					results := eval.Results{
						{
							Instance:    data.Labels{"instance": instanceLabel},
							State:       eval.Alerting,
							EvaluatedAt: now,
						},
					}
					fmt.Printf("EVAL done uid=%s scheduled=%s results=%d\n", ruleUID, now.Format(time.RFC3339), len(results))
					return results, nil
				},
			}

			evaluator := &fakeEvaluatorFactory{evaluator: fakeEvaluator}

			schedCfg := SchedulerCfg{
				MaxAttempts:      2,
				BaseInterval:     cfg.BaseInterval,
				C:                mockedClock,
				AppURL:           appUrl,
				EvaluatorFactory: evaluator,
				RuleStore:        ruleStore,
				Metrics:          testMetrics.GetSchedulerMetrics(),
				AlertSender:      notifier,
				Tracer:           testTracer,
				Log:              log.New("ngalert.scheduler"),
				RecordingWriter:  writer.NoopWriter{},
			}
			managerCfg := state.ManagerCfg{
				Metrics:       testMetrics.GetStateMetrics(),
				ExternalURL:   nil,
				InstanceStore: nil,
				Images:        &state.NoopImageService{},
				Clock:         mockedClock,
				Historian:     &state.FakeHistorian{},
				Tracer:        testTracer,
				Log:           log.New("ngalert.state.manager"),
			}
			st := state.NewManager(managerCfg, state.NewNoopPersister())
			sched := NewScheduler(schedCfg, st)

			evalAppliedCh := make(chan models.AlertRuleKey, 100)
			stopAppliedCh := make(chan models.AlertRuleKey, 10)

			sched.evalAppliedFunc = func(alertDefKey models.AlertRuleKey, now time.Time) {
				fmt.Printf("EVAL applied uid=%s at=%s\n", alertDefKey.UID, now.Format(time.RFC3339))
				select {
				case evalAppliedCh <- alertDefKey:
				default:
				}
			}
			sched.stopAppliedFunc = func(alertDefKey models.AlertRuleKey) {
				fmt.Printf("STOP applied uid=%s\n", alertDefKey.UID)
				select {
				case stopAppliedCh <- alertDefKey:
				default:
				}
			}

			withUID := func(uid string) models.AlertRuleMutator {
				return func(r *models.AlertRule) { r.UID = uid }
			}
			gen := models.RuleGen
			slowRule := gen.With(gen.WithOrgID(mainOrgID), gen.WithInterval(cfg.BaseInterval), withUID("a-slow")).GenerateRef()
			fastRule := gen.With(gen.WithOrgID(mainOrgID), gen.WithInterval(cfg.BaseInterval), withUID("b-fast")).GenerateRef()
			flakyRule := gen.With(gen.WithOrgID(mainOrgID), gen.WithInterval(cfg.BaseInterval), withUID("c-flaky")).GenerateRef()

			ruleStore.PutRule(ctx, slowRule, fastRule, flakyRule)

			fmt.Printf("SCENARIO start name=%s baseInterval=%s\n", name, cfg.BaseInterval)

			tick := mockedClock.Now().Add(cfg.BaseInterval).Truncate(cfg.BaseInterval)
			appliedCounts := make(map[string]int)
			scheduledCounts := make(map[string]int)

			for i := 1; i <= spec.ticks; i++ {
				mockedClock.Set(tick)
				readyToRun, stopped, _ := sched.processTick(ctx, dispatcherGroup, tick)

				readyUIDs := make([]string, 0, len(readyToRun))
				for _, item := range readyToRun {
					readyUIDs = append(readyUIDs, item.rule.UID)
					scheduledCounts[item.rule.UID]++
				}
				fmt.Printf("TICK %s index=%d ready=%v updated=0 stopped=%d\n", name, i, readyUIDs, len(stopped))

				if spec.applyRuleUpdate && i == 2 {
					fastRule.Version++
					ruleStore.PutRule(ctx, fastRule)
					fmt.Printf("RULE update uid=%s version=%d\n", fastRule.UID, fastRule.Version)
				}
				if spec.applyRuleUpdate && i == 3 {
					slowRule.Version++
					slowRule.Labels = map[string]string{"track": "slow", "updated": "true"}
					ruleStore.PutRule(ctx, slowRule)
					fmt.Printf("RULE update uid=%s version=%d labels=%v\n", slowRule.UID, slowRule.Version, slowRule.Labels)
				}
				if spec.deleteRuleMidway && i == 4 {
					deletedRules.Store(flakyRule.UID, true)
					ruleStore.DeleteRule(flakyRule)
					fmt.Printf("RULE delete uid=%s\n", flakyRule.UID)
				}

				time.Sleep(500 * time.Millisecond)
				tick = tick.Add(cfg.BaseInterval)
			}

			time.Sleep(2 * time.Second)

		drainApplied:
			for {
				select {
				case key := <-evalAppliedCh:
					appliedCounts[key.UID]++
				default:
					break drainApplied
				}
			}

			for uid, scheduled := range scheduledCounts {
				applied := appliedCounts[uid]
				dropped := scheduled - applied
				if dropped < 0 {
					dropped = 0
				}
				fmt.Printf("TALLY %s uid=%s scheduled=%d applied=%d dropped=%d\n", name, uid, scheduled, applied, dropped)
			}

			missedSlow := testutil.ToFloat64(testMetrics.GetSchedulerMetrics().EvaluationMissed.WithLabelValues(fmt.Sprint(mainOrgID), slowRule.Title))
			fmt.Printf("METRIC %s missed_evals_slow=%.0f\n", name, missedSlow)

			cancel()
			_ = dispatcherGroup.Wait()

		drainStop:
			for {
				select {
				case <-stopAppliedCh:
				case <-time.After(500 * time.Millisecond):
					break drainStop
				}
			}

			fmt.Printf("SCENARIO end name=%s\n", name)
		}

		reg := prometheus.NewRegistry()
		runScenario(scenario.name, scenario, reg)
	}
}

type fakeConditionEvaluator struct {
	evalFunc func(ctx context.Context, now time.Time, ruleUID string) (eval.Results, error)
	ruleUID  string
}

func (f *fakeConditionEvaluator) EvaluateRaw(ctx context.Context, now time.Time) (*backend.QueryDataResponse, error) {
	return nil, nil
}

func (f *fakeConditionEvaluator) Evaluate(ctx context.Context, now time.Time) (eval.Results, error) {
	return f.evalFunc(ctx, now, f.ruleUID)
}

type fakeEvaluatorFactory struct {
	evaluator *fakeConditionEvaluator
}

func (f *fakeEvaluatorFactory) Create(ctx eval.EvaluationContext, condition models.Condition) (eval.ConditionEvaluator, error) {
	uid := condition.Metadata["Uid"]
	return &fakeConditionEvaluator{evalFunc: f.evaluator.evalFunc, ruleUID: uid}, nil
}
```

The system decides what to work on next at the scheduler tick, and that choice first becomes visible as the ready set the scheduler returns for that tick. In the stress run, every tick printed a ready set via the runtime tick log, for example the first tick shows all three rules as ready, which is the earliest point in the flow where the decision appears: TICK stress index=1 ready=[a-slow b-fast c-flaky]. From there, each evaluation request is tied to a rule by the UID and the scheduled time, which appears in the EVAL start and EVAL done lines. Those two values let me match each response to the request even when completion timing varies, for example the flaky rule starts its first evaluation at scheduled 1970-01-01T00:00:01Z during tick 2 and is still applied with that scheduled time, as shown by EVAL done uid=c-flaky scheduled=1970-01-01T00:00:01Z and EVAL applied uid=c-flaky at=1970-01-01T00:00:01Z. That is the runtime signal for response matching, and it aligns with the scheduler and alert rule flow in schedule.go and alert_rule.go where evaluations are dispatched and processed by rule key.

Under stress, the timeout and retry behavior is explicit in the output. The slow rule shows a timeout error and a retry attempt for the same scheduled time: EVAL error uid=a-slow scheduled=1970-01-01T00:00:01Z err=context deadline exceeded is later followed by EVAL start uid=a-slow scheduled=1970-01-01T00:00:01Z call=2, and that second attempt completes and applies with the same scheduled time (EVAL done and EVAL applied at 1970-01-01T00:00:01Z). This indicates the system waits for the evaluation to complete, then retries when it gets a retryable error rather than bailing early, and it does so on the same scheduled tick, which is consistent with the retry loop in alert_rule.go where the attempt counter runs up to maxAttempts (set to 2 in this test) before giving up. The flaky rule's upstream timeout at call=2 was not retried: the rule had just been deleted, so its routine was stopped instead, which is visible as EVAL applied uid=c-flaky at=1970-01-01T00:00:02Z followed by STOP applied uid=c-flaky.

When evaluations fall behind, the coordination between scheduler send and rule evaluation receive shows up as dropped ticks rather than overlapping evaluations. The slow rule was scheduled on every tick, but the only EVAL start lines it produces in the stress run are for scheduled times 00:00:01Z and 00:00:05Z; the ticks for 00:00:02Z, 00:00:03Z, and 00:00:04Z never begin evaluating at all. In the code this is the send side in processTick handing the tick to the rule routine's Eval, which discards the pending evaluation when the routine is still busy; at that point schedule.go logs a "Tick dropped because alert rule evaluation is too slow" warning with the droppedTick value and increments the EvaluationMissed counter (schedule.go, around lines 378-380 at this commit). Grafana's infra logger is not attached to a console sink in this test binary, so the warning text itself does not appear in the captured output, but the counter does: METRIC stress missed_evals_slow=3 confirms exactly three ticks were discarded for the slow rule, matching the three missing scheduled times. The tally line TALLY stress uid=a-slow scheduled=5 applied=1 dropped=4 shows the backlog was discarded rather than queued; dropped is 4 rather than 3 at tally time because the 00:00:05Z evaluation was still in flight when the tally printed, and its EVAL applied line lands just after the METRIC line.

When a rule is removed while the system is under load, the runtime signs are visible in three places. After RULE delete uid=c-flaky on tick 4, the next tick's ready set shrinks and reports a stop: TICK stress index=5 ready=[a-slow b-fast] updated=0 stopped=1. STOP applied uid=c-flaky then confirms the rule's evaluation routine shut down cleanly, and no c-flaky line appears anywhere in the run after it. This is the deletion path where deleteAlertRule in schedule.go stops the routine with errRuleDeleted (defined as "rule deleted" in registry.go) and cleans up the rule's state cache. Nothing is left behind: the rule simply disappears from the ready set and from the evaluation stream.

A canceled evaluation looks clearly different from a normal completion. At the end of the stress scenario the slow rule's third call is interrupted mid-flight and prints EVAL canceled uid=a-slow scheduled=1970-01-01T00:00:05Z err=context canceled, where a normal completion prints EVAL done with results=1. The cancellation is followed by STOP applied uid=a-slow, showing the evaluation routine wound down cleanly instead of leaving anything running. So the distinguishing runtime signals are the context canceled error text in place of a done line with results, plus the STOP applied confirmation for the routine.

The answered versus unanswered tally under stress is visible at the end of the stress run. The slow rule shows TALLY stress uid=a-slow scheduled=5 applied=1 dropped=4, and the flaky rule shows TALLY stress uid=c-flaky scheduled=4 applied=2 dropped=2, while the healthy fast rule shows TALLY stress uid=b-fast scheduled=5 applied=5 dropped=0. Those counts change to all-applied under normal load, where every rule shows scheduled=4 applied=4 dropped=0 and METRIC normal missed_evals_slow=0. This directly demonstrates how the unanswered count increases when responses are delayed or dropped mid-burst in the stressed run and returns to zero in the normal run.

For out-of-order results, the evidence suggests ordering is preserved per rule even when completion timing varies. In the stress run, the slow rule only applies scheduled 1970-01-01T00:00:01Z and later 1970-01-01T00:00:05Z, with no applied results for 00:00:02Z through 00:00:04Z, which indicates drops rather than out-of-order application. In the normal run, the flaky rule applies 00:00:01Z, 00:00:02Z, 00:00:03Z, and 00:00:04Z in order even though its evaluations start about a tick behind the other rules, and there are no instances where a later scheduled time is applied before an earlier one for the same UID. The observable behavior that suggests ordering is preserved is the monotonic scheduled timestamps in the EVAL applied lines per rule, together with the drop signals (missing scheduled times and the nonzero missed counter) whenever the system chooses to skip work instead of reordering it.

The most obvious rhythm change between stressed and normal runs is the presence of timeout errors and dropped work in the stressed run, and their absence in the normal run. Under stress, there is a long gap between EVAL applied lines for a-slow (nothing applies between 00:00:01Z and 00:00:05Z), error lines for context deadline exceeded and upstream timeout appear, and the missed counter ends at 3. Under normal load, evaluations start and complete for each tick without errors, the volume of output is steady and evenly spaced per tick, and all scheduled evaluations are applied.

The temporary test file was removed after the run and git status confirms the repository was left unchanged.
